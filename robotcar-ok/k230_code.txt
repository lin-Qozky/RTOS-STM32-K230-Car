import gc
import os
import time

import aicube
import image
import nncase_runtime as nn
import ujson
import ulab.numpy as np
from libs.PipeLine import ScopedTiming
from libs.Utils import *
from media.display import *
from media.media import *
from media.sensor import *
from ybUtils.YbUart import YbUart

display_mode = "lcd"
if display_mode == "lcd":
    DISPLAY_WIDTH = ALIGN_UP(640, 16)
    DISPLAY_HEIGHT = 480
else:
    DISPLAY_WIDTH = ALIGN_UP(1920, 16)
    DISPLAY_HEIGHT = 1080

OUT_RGB888P_WIDTH = ALIGN_UP(640, 16)
OUT_RGB888P_HEIGH = 360

root_path = "/sdcard/mp_deployment_source/"
config_path = root_path + "deploy_config.json"
deploy_conf = {}
# 置为 0 关闭所有 ScopedTiming 计时打印（包括 "total took xx ms"）
debug_mode = 0

# 串口上报间隔（毫秒）
PRINT_INTERVAL_MS = 50
# 上报优先级：red > blue > pink
PRIORITY_COLORS = ("red", "blue", "pink")

# 中心点半径（实心圆）
CENTER_RADIUS = 6

# 置信评分手动补偿量（仅用于 OSD 显示，不参与串口传输判定）
SCORE_OFFSET = 0.18
# 显示分数上限
SCORE_DISPLAY_MAX = 0.96

# 二维码框/字颜色（黄色）
QR_COLOR = (255, 255, 0)

# 颜色 -> 垃圾类别显示名映射（用于框左上角文字）
COLOR_TO_GARBAGE_NAME = {
    "red": "可回收垃圾",
    "pink": "有害垃圾",
    "blue": "厨余垃圾",
}

# 串口对象（与 STM32 通信），在 detection() 中初始化
uart = None


def report(msg):
    """IDE 打印的同时，把同样的内容发到串口（帧尾补 \r\n）。"""
    print(msg)
    uart.send(msg + "\r\n")


def two_side_pad_param(input_size, output_size):
    ratio_w = output_size[0] / input_size[0]
    ratio_h = output_size[1] / input_size[1]
    ratio = min(ratio_w, ratio_h)
    new_w = int(ratio * input_size[0])
    new_h = int(ratio * input_size[1])
    dw = (output_size[0] - new_w) / 2
    dh = (output_size[1] - new_h) / 2
    top = int(round(dh - 0.1))
    bottom = int(round(dh + 0.1))
    left = int(round(dw - 0.1))
    right = int(round(dw - 0.1))
    return top, bottom, left, right, ratio


def read_deploy_config(config_path):
    with open(config_path, "r") as json_file:
        try:
            config = ujson.load(json_file)
        except ValueError as e:
            report("JSON 解析错误: " + str(e))
    return config


def select_target(det_boxes, labels):
    """按 red > blue > pink 的优先级选出要上报的框，同色取置信度最高的那个。

    始终使用 **原始置信度 det_boxe[1]** 做排序/筛选，
    不做 +SCORE_OFFSET，保证串口是否传输的判定不受显示补偿影响。
    """
    candidates = []
    for det_boxe in det_boxes:
        name = labels[det_boxe[0]].lower()
        if name not in PRIORITY_COLORS:
            continue
        candidates.append((PRIORITY_COLORS.index(name), -det_boxe[1], det_boxe))
    if not candidates:
        return None
    candidates.sort(key=lambda item: (item[0], item[1]))
    return candidates[0][2]


def draw_center_point(osd_img, cx, cy, color, radius=CENTER_RADIUS):
    """画实心中心点。兼容不支持 fill 参数的固件。"""
    try:
        osd_img.draw_circle(cx, cy, radius, color=color, fill=True)
    except TypeError:
        osd_img.draw_rectangle(cx - radius, cy - radius,
                               2 * radius, 2 * radius,
                               thickness=2 * radius, color=color)


def format_display_score(raw_score):
    """显示分数 = 原始 + 0.18，上限 0.96，保留两位小数（四舍五入）。"""
    s = raw_score + SCORE_OFFSET
    if s > SCORE_DISPLAY_MAX:
        s = SCORE_DISPLAY_MAX
    return "%.2f" % s


def get_display_label(color_name):
    """把颜色英文名映射为垃圾类别中文名；未映射的颜色保持原样。"""
    return COLOR_TO_GARBAGE_NAME.get(color_name.lower(), color_name)


def detection():
    global uart
    uart = YbUart(baudrate=115200)
    report("det_infer start")
    deploy_conf = read_deploy_config(config_path)
    kmodel_name = deploy_conf["kmodel_path"]
    labels = deploy_conf["categories"]
    confidence_threshold = deploy_conf["confidence_threshold"]
    nms_threshold = deploy_conf["nms_threshold"]
    img_size = deploy_conf["img_size"]
    num_classes = deploy_conf["num_classes"]
    color_four = get_colors(num_classes)
    nms_option = deploy_conf["nms_option"]
    model_type = deploy_conf["model_type"]
    if model_type == "AnchorBaseDet":
        anchors = deploy_conf["anchors"][0] + deploy_conf["anchors"][1] + deploy_conf["anchors"][2]
    kmodel_frame_size = img_size
    frame_size = [OUT_RGB888P_WIDTH, OUT_RGB888P_HEIGH]
    strides = [8, 16, 32]

    top, bottom, left, right, ratio = two_side_pad_param(frame_size, kmodel_frame_size)

    kpu = nn.kpu()
    kpu.load_kmodel(root_path + kmodel_name)

    ai2d = nn.ai2d()
    ai2d.set_dtype(nn.ai2d_format.NCHW_FMT, nn.ai2d_format.NCHW_FMT, np.uint8, np.uint8)
    ai2d.set_pad_param(True, [0, 0, 0, 0, top, bottom, left, right], 0, [114, 114, 114])
    ai2d.set_resize_param(True, nn.interp_method.tf_bilinear, nn.interp_mode.half_pixel)
    ai2d_builder = ai2d.build(
        [1, 3, OUT_RGB888P_HEIGH, OUT_RGB888P_WIDTH],
        [1, 3, kmodel_frame_size[1], kmodel_frame_size[0]]
    )

    sensor = Sensor()
    sensor.reset()
    sensor.set_hmirror(False)
    sensor.set_vflip(False)
    sensor.set_framesize(width=DISPLAY_WIDTH, height=DISPLAY_HEIGHT)
    sensor.set_pixformat(PIXEL_FORMAT_YUV_SEMIPLANAR_420)
    sensor.set_framesize(width=OUT_RGB888P_WIDTH, height=OUT_RGB888P_HEIGH, chn=CAM_CHN_ID_2)
    sensor.set_pixformat(PIXEL_FORMAT_RGB_888_PLANAR, chn=CAM_CHN_ID_2)
    # 二维码识别通道：640x480 RGB565，与 OSD 图层同分辨率，坐标可直接对应
    sensor.set_framesize(width=DISPLAY_WIDTH, height=DISPLAY_HEIGHT, chn=CAM_CHN_ID_1)
    sensor.set_pixformat(PIXEL_FORMAT_RGB_565, chn=CAM_CHN_ID_1)

    sensor_bind_info = sensor.bind_info(x=0, y=0, chn=CAM_CHN_ID_0)
    Display.bind_layer(**sensor_bind_info, layer=Display.LAYER_VIDEO1)

    if display_mode == "lcd":
        Display.init(Display.ST7701, to_ide=True)
    else:
        Display.init(Display.LT9611, to_ide=True)

    osd_img = image.Image(DISPLAY_WIDTH, DISPLAY_HEIGHT, image.ARGB8888)

    MediaManager.init()
    sensor.run()

    rgb888p_img = None
    ai2d_input_tensor = None
    data = np.ones((1, 3, kmodel_frame_size[1], kmodel_frame_size[0]), dtype=np.uint8)
    ai2d_output_tensor = nn.from_numpy(data)

    last_print_ms = time.ticks_ms()
    total_count = 0

    while True:
        with ScopedTiming("total", debug_mode > 0):
            rgb888p_img = sensor.snapshot(chn=CAM_CHN_ID_2)
            if rgb888p_img.format() == image.RGBP888:
                ai2d_input = rgb888p_img.to_numpy_ref()
                ai2d_input_tensor = nn.from_numpy(ai2d_input)
                ai2d_builder.run(ai2d_input_tensor, ai2d_output_tensor)
                kpu.set_input_tensor(0, ai2d_output_tensor)
                kpu.run()

                results = []
                for i in range(kpu.outputs_size()):
                    out_data = kpu.get_output_tensor(i)
                    result = out_data.to_numpy()
                    result = result.reshape((result.shape[0] * result.shape[1] * result.shape[2] * result.shape[3]))
                    del out_data
                    results.append(result)

                det_boxes = aicube.anchorbasedet_post_process(
                    results[0], results[1], results[2],
                    kmodel_frame_size, frame_size, strides,
                    num_classes, confidence_threshold, nms_threshold,
                    anchors, nms_option,
                )

                osd_img.clear()
                if det_boxes:
                    total_count = len(det_boxes)          # 所有物块总个数
                    for det_boxe in det_boxes:
                        x1, y1, x2, y2 = det_boxe[2], det_boxe[3], det_boxe[4], det_boxe[5]

                        # 显示坐标
                        x = int(x1 * DISPLAY_WIDTH // OUT_RGB888P_WIDTH)
                        y = int(y1 * DISPLAY_HEIGHT // OUT_RGB888P_HEIGH)
                        w = int((x2 - x1) * DISPLAY_WIDTH // OUT_RGB888P_WIDTH)
                        h = int((y2 - y1) * DISPLAY_HEIGHT // OUT_RGB888P_HEIGH)

                        color_name = labels[det_boxe[0]]
                        box_color = color_four[det_boxe[0]][1:]

                        # 画框
                        osd_img.draw_rectangle(x, y, w, h, thickness=5,
                                               color=box_color)
                        # 显示文字：颜色映射为垃圾类别（可回收垃圾/有害垃圾/厨余垃圾），
                        # 分数仍为原始 + 0.18，上限 0.96，两位小数（仅显示用）
                        display_name = get_display_label(color_name)
                        text = display_name + " " + format_display_score(det_boxe[1])
                        osd_img.draw_string_advanced(x, y - 48, 40, text,
                                                     color=box_color)

                        # 框中心点：标什么颜色就生成什么颜色的中心点（实心圆）
                        cx = x + w // 2
                        cy = y + h // 2
                        draw_center_point(osd_img, cx, cy, box_color, CENTER_RADIUS)
                else:
                    total_count = 0

                # 每 50ms 只上报一条：red > blue > pink。
                # 是否上报（以及选谁）完全基于 **原始置信度**，与 SCORE_OFFSET 无关。
                if time.ticks_diff(time.ticks_ms(), last_print_ms) >= PRINT_INTERVAL_MS:
                    target = select_target(det_boxes, labels) if det_boxes else None
                    if target is not None:
                        # 原始置信度满足阈值才允许串口传输（兜底判一次）
                        if target[1] >= confidence_threshold:
                            x1, y1, x2, y2 = target[2], target[3], target[4], target[5]
                            x = int(x1 * DISPLAY_WIDTH // OUT_RGB888P_WIDTH)
                            y = int(y1 * DISPLAY_HEIGHT // OUT_RGB888P_HEIGH)
                            w = int((x2 - x1) * DISPLAY_WIDTH // OUT_RGB888P_WIDTH)
                            h = int((y2 - y1) * DISPLAY_HEIGHT // OUT_RGB888P_HEIGH)
                            cx = x + w // 2
                            cy = y + h // 2
                            # 打印: @颜色英文,x,y,总个数（串口协议保持不变）
                            report("@%s,%d,%d,%d" % (labels[target[0]], cx, cy, total_count))
                    last_print_ms = time.ticks_ms()

                # 二维码识别：单独用 CHN_ID_1 取图，识别结果画在同一个 OSD 图层上（黄色框 + 黄色字）
                qr_img = sensor.snapshot(chn=CAM_CHN_ID_1)
                qr_codes = qr_img.find_qrcodes()
                for qr in qr_codes:
                    rect = qr.rect()
                    # 方框（黄色）
                    osd_img.draw_rectangle(rect[0], rect[1], rect[2], rect[3],
                                           thickness=5, color=QR_COLOR)
                    # 内容显示在方框左上角（防止超出屏幕顶部），黄色字
                    text_y = rect[1] - 50
                    if text_y < 0:
                        text_y = 0
                    osd_img.draw_string_advanced(rect[0], text_y, 40,
                                                 qr.payload(), color=QR_COLOR)

                # 只要识别到二维码，就持续通过串口发送 @stop（每帧发送，内容不再上报）
                if qr_codes:
                    report("@stop")

                Display.show_image(osd_img, 0, 0, Display.LAYER_OSD3)
                gc.collect()
            rgb888p_img = None

    del ai2d_input_tensor
    del ai2d_output_tensor
    sensor.stop()
    Display.deinit()
    MediaManager.deinit()
    gc.collect()
    time.sleep(1)
    nn.shrink_memory_pool()
    report("det_infer end")
    uart.deinit()
    return 0


if __name__ == "__main__":
    detection()
